# Runtime dependencies for the University IT Service Desk assistant.
#
# The system deliberately needs very little. Only langchain-core is imported by
# the code; everything else (HTTP, tokenisation, BM25, scoring) is standard
# library. There is no vector database and no embedding model, because the
# knowledge base is small, dense and highly repetitive, and because no embedding
# model is available locally.

langchain-core>=1.0,<2.0

# ---------------------------------------------------------------------------
# Optional / development only -- NOT required to run the agent.
#
#   langchain                >=1.3     (convenience imports, not used)
#   langchain-community      >=0.4     (not used: it no longer exports ChatOllama)
#   chromadb                 >=1.0     (not used: retrieval is BM25 in rag.py)
#   langgraph                >=1.2     (not used: orchestration is in agent.py)
#
# They are installed in the development environment and the code runs alongside
# them, but `pip install langchain-core` is sufficient for a clean checkout.
# ---------------------------------------------------------------------------
